  The client should send only 24h, not its own start and end dates. This keeps the calculation consistent and prevents manipulated
  ranges.

  ## Calculate the 24-hour range

  function getLast24HourRange(now = new Date()) {
    // End of the current hourly bucket.
    const rangeEnd = new Date(now);
    rangeEnd.setUTCMinutes(0, 0, 0);
    rangeEnd.setUTCHours(rangeEnd.getUTCHours() + 1);

    // Exactly 24 hourly buckets before rangeEnd.
    const rangeStart = new Date(rangeEnd);
    rangeStart.setUTCHours(rangeStart.getUTCHours() - 24);

    return {
      rangeStart,
      rangeEnd,
    };
  }

  If the current time is:

  2026-07-23 14:37 UTC

  the calculated range is:

  rangeStart: 2026-07-22 15:00 UTC
  rangeEnd:   2026-07-23 15:00 UTC

  This represents 24 chart buckets, including the current incomplete hour.

  ## Query the payments

  Use receivedAmount for accuracy. Only use paymentIntent.amount if underpayment and overpayment cannot happen.

  ## Create 24 empty buckets

  Creating empty buckets is important. Otherwise, hours without payments will disappear from the graph.

  type HourlyBucket = {
    periodStart: Date;
    periodEnd: Date;
    grossVolume: Prisma.Decimal;
    fees: Prisma.Decimal;
    refunds: Prisma.Decimal;
    netVolume: Prisma.Decimal;
    paymentCount: number;
  };

  function createHourlyBuckets(
    rangeStart: Date,
    numberOfHours: number,
  ) {
    const buckets = new Map<string, HourlyBucket>();

    for (let index = 0; index < numberOfHours; index++) {
      const periodStart = new Date(rangeStart);
      periodStart.setUTCHours(periodStart.getUTCHours() + index);

      const periodEnd = new Date(periodStart);
      periodEnd.setUTCHours(periodEnd.getUTCHours() + 1);

      buckets.set(periodStart.toISOString(), {
        periodStart,
        periodEnd,
        grossVolume: new Prisma.Decimal(0),
        fees: new Prisma.Decimal(0),
        refunds: new Prisma.Decimal(0),
        netVolume: new Prisma.Decimal(0),
        paymentCount: 0,
      });
    }

    return buckets;
  }

  ## Put payments into their hourly buckets

  const buckets = createHourlyBuckets(rangeStart, 24);

  for (const payment of payments) {
    if (!payment.confirmedAt) continue;

    const { periodStart } =
      getHourlyPeriod(payment.confirmedAt);

    const bucket = buckets.get(periodStart.toISOString());

    if (!bucket) continue;

    const grossAmount = payment.receivedAmount;
    const feeAmount = payment.feeAmount;
    const netAmount = grossAmount.minus(feeAmount);

    bucket.grossVolume =
      bucket.grossVolume.plus(grossAmount);

    bucket.fees =
      bucket.fees.plus(feeAmount);

    bucket.netVolume =
      bucket.netVolume.plus(netAmount);

    bucket.paymentCount += 1;
  }

  The hourly helper is:

  function getHourlyPeriod(dateInput: Date | string) {
    const date = new Date(dateInput);

    if (Number.isNaN(date.getTime())) {
      throw new Error("Invalid date");
    }

    const periodStart = new Date(date);
    periodStart.setUTCMinutes(0, 0, 0);

    const periodEnd = new Date(periodStart);
    periodEnd.setUTCHours(periodEnd.getUTCHours() + 1);

    return {
      periodStart,
      periodEnd,
    };
  }
